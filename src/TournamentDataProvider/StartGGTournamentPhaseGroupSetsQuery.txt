query TournamentPhaseGroupSetsQuery($id: ID!, $page: Int!, $perPage: Int!) {
	phaseGroup(id: $id) {
		sets(page: $page, perPage: $perPage, filters: {showByes: true, hideEmpty: false}, sortType: CALL_ORDER) {
			pageInfo {
				totalPages
			}
			nodes {
				id
				round
				identifier
				fullRoundText
				entrant1Score
				entrant2Score
				winnerId
				state
				slots {
					prereqType
					prereqId
					prereqPlacement
					seed {
						id
					}
					entrant {
						id
					}
				}
			}
		}
	}
}
